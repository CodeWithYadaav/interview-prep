function throwsErrorRandomly() {
  if (Math.random() > 0.1) {
    throw new Error('Bad Luck!');
  }
  return 'Good Luck!';
}

async function retry(fn, maxNumberOfRetries, delayBetweenRetry) {

  for (let attempt = 1; attempt <= maxNumberOfRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (attempt === maxNumberOfRetries) {
        throw error;
      }
      console.log(
        `Attempt ${attempt} failed. Retrying  in ${delayBetweenRetry} ms...`
      );
      await new Promise(resolve => setTimeout(resolve, delayBetweenRetry)); 
    }
  }
}

// Usage
retry(throwsErrorRandomly, 3, 500)
  .then(console.log)
  .catch(error => console.error('All attempts failed:', error.message));
