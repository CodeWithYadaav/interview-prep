1. Retrieve the total number of orders placed by each customer. 
 SELECT customer_id, COUNT(order_id) AS total_orders FROM Orders GROUP BY customer_id;

2. Get the average salary of employees by department. 
 SELECT department, AVG(salary) AS average_salary FROM Employee GROUP BY department;

3. List all employees who were hired in 2023. 
 SELECT * FROM Employee WHERE YEAR(hire_date) = 2023;

4. Find the highest sales amount recorded for each product. 
 SELECT product_id, MAX(sales_amount) AS highest_sales FROM Sales GROUP BY product_id;

5. Retrieve customers who have made purchases greater than $500. 
 SELECT customer_id FROM Orders WHERE total_amount > 500;

6. List products that have never been sold. 
 SELECT p.product_id, p.product_name FROM Products p LEFT JOIN Sales s ON p.product_id = s.product_id WHERE s.product_id IS NULL;

7. Get the count of distinct products sold each month. 
 SELECT MONTH(order_date) AS month, COUNT(DISTINCT product_id) AS distinct_products FROM Sales GROUP BY MONTH(order_date);

8. Find the employee(s) with the lowest salary.
 SELECT * FROM Employee WHERE salary = (SELECT MIN(salary) FROM Employee);

9. Retrieve all customers and their corresponding orders. 
 SELECT c.customer_id, o.order_id FROM Customers c LEFT JOIN Orders o ON c.customer_id = o.customer_id;

10. Calculate the total revenue generated by each product.
 SELECT product_id, SUM(quantity * price) AS total_revenue FROM Sales GROUP BY product_id;


11. Get top 3 best-selling products by revenue
 SELECT product_id, SUM(quantity * price) AS revenue
FROM Sales
GROUP BY product_id
ORDER BY revenue DESC
LIMIT 3;




12. Find customers who placed orders in two or more consecutive months
SELECT DISTINCT customer_id
FROM (
  SELECT customer_id, MONTH(order_date) AS month, 
         LAG(MONTH(order_date)) OVER (PARTITION BY customer_id ORDER BY order_date) AS prev_month
  FROM Orders
) t
WHERE month = prev_month + 1;





13. Detect duplicate records in a table
SELECT email, COUNT(*) 
FROM Users 
GROUP BY email 
HAVING COUNT(*) > 1;





14. Retrieve nth highest salary (e.g., 3rd highest)
SELECT DISTINCT salary
FROM Employee e1
WHERE 3 = (
  SELECT COUNT(DISTINCT salary)
  FROM Employee e2
  WHERE e2.salary >= e1.salary
);




15. Show products ordered more than once by the same customer
SELECT customer_id, product_id, COUNT(*) AS order_count
FROM Orders
GROUP BY customer_id, product_id
HAVING COUNT(*) > 1;





16. List customers who haven’t ordered in the last 6 months
SELECT c.customer_id
FROM Customers c
LEFT JOIN Orders o ON c.customer_id = o.customer_id AND o.order_date > NOW() - INTERVAL 6 MONTH
WHERE o.order_id IS NULL;




17. Calculate running total of daily sales
SELECT order_date, 
       SUM(quantity * price) OVER (ORDER BY order_date) AS running_total
FROM Sales;





18. Rank employees by salary within each department
SELECT employee_id, department, salary,
       RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS dept_rank
FROM Employee;





19. Retrieve departments where average salary > global average salary
SELECT department
FROM Employee
GROUP BY department
HAVING AVG(salary) > (SELECT AVG(salary) FROM Employee);





20. Get customer details with their most recent order
SELECT c.customer_id, c.name, o.order_id, o.order_date
FROM Customers c
JOIN Orders o ON c.customer_id = o.customer_id
WHERE o.order_date = (
  SELECT MAX(order_date) 
  FROM Orders 
  WHERE customer_id = c.customer_id
);

| employee\_id | name    | manager\_id |
| ------------ | ------- | ----------- |
| 1            | Alice   | NULL        |
| 2            | Bob     | 1           |
| 3            | Charlie | 1           |
| 4            | David   | 2           |



SELECT 
    e.name AS employee_name,
    m.name AS manager_name
FROM 
    Employee e
LEFT JOIN 
    Employee m ON e.manager_id = m.employee_id;




✅ What is a Stored Procedure (SP)?
A Stored Procedure is a precompiled collection of one or more SQL statements that you can execute as a single unit. It is stored in the database and can be reused multiple times.

CREATE PROCEDURE GetEmployeeByDept(IN dept_name VARCHAR(50))
BEGIN
    SELECT * FROM Employees WHERE department = dept_name;
END;



To call this
CALL GetEmployeeByDept('HR');


| Benefit                   | Description                                               |
| ------------------------- | --------------------------------------------------------- |
| ✅ Reusability             | Write once, call many times.                              |
| ✅ Security                | Can restrict direct access to tables and expose only SPs. |
| ✅ Performance             | Precompiled, so faster execution.                         |
| ✅ Maintenance             | Business logic encapsulated in DB, easier to manage.      |
| ✅ Reduced network traffic | Send a single `CALL` instead of multiple SQLs.            |


22.✅ What is a View in SQL?
A View is a virtual table based on the result of a SELECT query. It doesn’t store data itself, but rather shows data from one or more tables.

| Concept                   | Description                                                     |
| ------------------------- | --------------------------------------------------------------- |
| 🔍 Virtual Table          | Acts like a table, but stores **no data**.                      |
| 📄 Read-Only or Updatable | Depends on the query logic.                                     |
| 🔄 Dynamic                | Reflects changes made to the underlying tables.                 |
| 📚 Used for               | Simplifying complex queries, data abstraction, security, reuse. |


-- Create a simple view
CREATE VIEW EmployeeView AS
SELECT name, department, salary
FROM Employees
WHERE status = 'Active';
