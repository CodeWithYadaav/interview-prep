//Square of numbers in ascending way
function square(arr) {
    let result = []
    for (let i = 0; i < arr.length; i++) {
        result.push(arr[i] * arr[i])
    }
    result.sort()
    return result
}


const res = [-4, -1, 0, 3, 10]
console.log(square(res))

function maxOccurence(str) {
    let count = {}
    let maxCount = 0;
    let maxChar = ''
    for (let char of str) {
        count[char] = (count[char] || 0) + 1
    }

    for (let char in count) {

//Dupliacte on sorted array 
Remove Duplicates from Sorted Array
function duplaicate(arr) {
    let left = 0;
    for (let right = 1; right < arr.length; right++) {
        if (arr[left] != arr[right]) {
            left++
            arr[left] = arr[right]
        }
    }
    return left + 1
}

//duplicate on unsorted array
function dups(nums) {
    let res = []
    let seen = {}

    for (let num of nums) {
        if (seen[num]) {
            if (!res.includes(num)) {
                res.push(num)
            }
        } else {
            seen[num] = true
        }
    }
    return res
}


const nums = [1, 1, 2, 3, 3]
console.log(dups(nums))








//Anagram
function anagram(str1, str2) {

    let count = {}
    for (let char of str1) {
        count[char] = (count[char] || 0) + 1
    }

    for (let char of str2) {
        if (!count[char]) {
            return 'not valid'
        }
        count[char]--
    }
    return 'is valid'
}


const str1 = 'tar'
const str2 = 'aat'
console.log(anagram(str1, str2))








//2 sum using 2 pointers on sorted array 
function twoSum(arr, target) {
    let left = 0
    let right = arr.length - 1
    while (left < right) {
        const sum = arr[left] + arr[right]
        if (sum == target) {
            return [arr[left], arr[right]]
        } else if (sum < target) {
            left++
        } else { right-- }
    }
}




const num = [1, 2, 3, 4]
console.log(twoSum(num, 5))










//productItself using prefix and suffix 
function productItself(arr) {
    let res = new Array(arr.length).fill(1)
    let prefix = 1
    for (let i = 0; i < arr.length; i++) {
        res[i] = prefix
        prefix *= arr[i]
    }
    let suffix = 1;
    for (let i = arr.length - 1; i >= 0; i--) {
        res[i] *= suffix
        suffix *= arr[i]
    }
    return res
}

const num = [1, 2, 3, 4]
console.log(productItself(num))









//alternateUpperCase
function toAlternateUpperCase(str) {
    let res = str.split('')
    for (let i = 0; i < res.length; i++) {
        if (i % 2 == 0) {
            res[i] = res[i].toLowerCase()
        } else {
            res[i] = res[i].toUpperCase()
        }
    }
    return res.join('')
}

const str = 'praveen'
console.log(toAlternateUpperCase(str))










//reverse a string 
function reverse(str) {
    const arr = str.split('')
    console.log(arr)
    let left = 0;
    let right = arr.length - 1
    while (left < right) {
        [arr[left], arr[right]] = [arr[right], arr[left]]
        left++
        right--
    }
    return arr.join('')
}

const res = 'hello'
console.log(reverse(res))




//valid palindrome 
function isPalindrome(s) {
    let left = 0;
    let right = s.length - 1;

    while (left < right) {
        if (s[left].toLowerCase() !== s[right].toLowerCase()) {
            return false;
        }
        left++;
        right--;
    }

    return true;
}

console.log(isPalindrome("madam")); // true
console.log(isPalindrome("racecar")); // true
console.log(isPalindrome("hello")); // false








//3 sum
function threeSum(arr) {
    const res = []
    const nums = arr.sort((a, b) => a - b)

    for (let i = 0; i < nums.length - 2; i++) {
        if (i > 0 && nums[i] == nums[i - 1]) continue

        let left = i + 1;
        let right = nums.length - 1
        while (left < right) {
            const sum = nums[i] + nums[left] + nums[right]

            if (sum == 0) {
                res.push([nums[i], nums[left], nums[right]])

                while (nums[left] == nums[left + 1]) left++
                while (nums[right] == nums[right - 1]) right--
                left++
                right--
            } else if (sum < 0) {
                left++
            } else {
                right--
            }

        }
    }
    return res

}


const arr = [-1, 0, 1, 2, -1, -4];
console.log(threeSum(arr))










//Flaten an array
function flaten(arr) {
    let res = []
    for (let item of arr) {
        if (Array.isArray(item)) {
            res = res.concat(flaten(item))
        } else {
            res.push(item)
        }
    }
    return res
}


const arr = [1, [2, [3, 4], 5], 6];
console.log(flaten(arr))














//secondHighest from array 
function secondHighest(arr) {
    let first = -Infinity
    let second = -Infinity

    for (let num of arr) {
        if (num > first) {
            second = first
            first = num
        } else if (num > second && num !== first) {
            second = num
        }
    }
    return second == -Infinity ? null : second
}



const arr = [10, 5, 8, 20, 9, 20]
console.log(secondHighest(arr))







//sort an array (using bubble but not higly optimised) can go with merge sort
function bubbleSort(arr) {

    for (let i = 0; i < arr.length; i++) {
        for (let j = 0; j < arr.length - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]]
            }
        }
    }
    return arr

}


const arr = [5, 3, 8, 4, 2];
console.log(bubbleSort(arr));
