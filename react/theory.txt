/* ============================================================================
 * 1. CLOSURES IN JAVASCRIPT & REACT
 * ============================================================================
 *
 * 🎯 ONE-LINE INTERVIEW ANSWER:
 * "A closure is a function that retains access to its lexical scope (outer variables)
 *  even after the outer function has finished executing."
 *
 * 💡 REACT RELEVANCE:
 *  React Hooks (`useState`, `useEffect`, `useCallback`) rely directly on closures
 *  to preserve state across render cycles.
 */

// Example: Function Factory & State Encapsulation
function createCounter(initialValue = 0) {
  let count = initialValue; // Private variable enclosed within inner functions

  return {
    increment: () => ++count,
    decrement: () => --count,
    getValue: () => count,
  };
}

const counter = createCounter(10);
console.log(counter.increment()); // 11
console.log(counter.getValue());   // 11 (Accesses `count` via closure)

/* ============================================================================
 * 2. PROPS & PROP DRILLING
 * ============================================================================
 *
 * 🎯 ONE-LINE INTERVIEW ANSWER:
 * "Props are immutable read-only inputs passed from parent to child components, whereas
 *  Prop Drilling occurs when props are forced through intermediate components that don't
 *  need them just to reach a deep child."
 *
 * 🚨 PROBLEMS WITH PROP DRILLING:
 *  1. Code Clutter: Bloats intermediate components with unused prop interfaces.
 *  2. Refactoring Overhead: Renaming or changing a prop type requires edits across N files.
 *  3. Unnecessary Re-renders: Intermediate wrappers re-render on prop updates unless memoized.
 */

// ❌ ANTI-PATTERN: Prop Drilling
function Grandparent() {
  const user = { name: "Praveen" };
  return <Parent user={user} />;
}

function Parent({ user }) {
  // Parent doesn't use `user`, but must pass it down
  return <Child user={user} />;
}

function Child({ user }) {
  return <h1>Hello, {user.name}</h1>;
}

// ✅ SOLUTION 1: React Context API
const UserContext = React.createContext(null);

function ContextGrandparent() {
  const user = { name: "Praveen" };
  return (
    <UserContext.Provider value={user}>
      <ContextParent />
    </UserContext.Provider>
  );
}

function ContextParent() {
  return <ContextChild />; // Clean intermediate tree!
}

function ContextChild() {
  const user = React.useContext(UserContext); // Consumes directly
  return <h1>Hello, {user.name}</h1>;
}

/*
 * 🛠️ ALTERNATIVE SOLUTIONS TO PROP DRILLING:
 *  - Component Composition: Pass children directly (`<Parent><Child user={user} /></Parent>`).
 *  - State Management Libraries: Redux Toolkit, Zustand, or Recoil for global/complex state.
 */

/* ============================================================================
 * 4. PERFORMANCE OPTIMIZATION: React.memo vs useMemo vs useCallback
 * ============================================================================
 *
 * ┌──────────────────┬───────────────────────────────┬──────────────────────────────────┐
 * │ Feature          │ Target                        │ Primary Purpose                  │
 * ├──────────────────┼───────────────────────────────┼──────────────────────────────────┤
 * │ React.memo       │ Component (HOC)               │ Prevents component re-renders    │
 * │                  │                               │ if props haven't changed         │
 * │ useMemo          │ Value (Calculation result)    │ Caches expensive computations    │
 * │ useCallback      │ Function reference            │ Prevents function re-creation    │
 * │                  │                               │ across render cycles             │
 * └──────────────────┴───────────────────────────────┴──────────────────────────────────┘
 */

// Example demonstrating all three optimization tools together:

const ExpensiveChild = React.memo(function Child({ onClick, computedValue }) {
  console.log("Child rendered!");
  return <button onClick={onClick}>Value: {computedValue}</button>;
});

function ParentDashboard({ count, text }) {
  // 1. useMemo: Avoid recalculating prime numbers or sorting large arrays on every render
  const expensiveCalculation = React.useMemo(() => {
    return count * 1000; // Simulated heavy calculation
  }, [count]);

  // 2. useCallback: Keep the exact same function memory reference between renders
  //    so `React.memo` inside `ExpensiveChild` doesn't see a "new" prop reference.
  const handleClick = React.useCallback(() => {
    console.log("Button clicked!");
  }, []);

  return (
    <div>
      <input value={text} onChange={(e) => setText(e.target.value)} />
      <ExpensiveChild onClick={handleClick} computedValue={expensiveCalculation} />
    </div>
  );
}

/* ============================================================================
 * 6. SYSTEM DESIGN: CACHE INVALIDATION & RELEASING NEW BUILDS TO CDN
 * ============================================================================
 *
 * ❓ Interviewer Question:
 * "When you deploy a new React build to S3/CloudFront, how do you ensure users instantly
 *  get the latest JS/CSS files instead of stale cached versions?"
 *
 * 💡 SENIOR ARCHITECT ANSWER:
 * "We use Cache-Busting File Hashing paired with selective HTTP Cache-Control headers."
 *
 * 🛠️ IMPLEMENTATION STRATEGY:
 *
 * 1. Asset Cache-Busting (Hashed Filenames):
 *    - Modern bundlers append unique content hashes to output files: `main.a8f93c2.js`.
 *    - When code changes, the filename changes. CloudFront sees it as a completely new asset
 *      and fetches it immediately from the origin.
 *
 * 2. Cache-Control Header Strategy at Origin (S3):
 *    - Hashed Assets (`/static/*.js`, `/static/*.css`):
 *      Header: `Cache-Control: public, max-age=31536000, immutable`
 *      (Safely cached at CDN and browser for 1 year).
 *
 *    - Entry Point (`index.html`):
 *      Header: `Cache-Control: no-cache, no-store, must-revalidate`
 *      (Forces the browser to always revalidate `index.html` on every page load).
 *
 * 3. CDN Cache Invalidation (Fallback / Hotfixes):
 *    - For un-hashed emergency fixes or `index.html`, trigger an explicit CloudFront Invalidation
 *      via CI/CD pipeline (`aws cloudfront create-invalidation --distribution-id XYZ --paths "/*"`).
 */
