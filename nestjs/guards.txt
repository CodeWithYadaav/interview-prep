// ✅ What is a Guard in NestJS?
// A Guard in NestJS is a class used to control whether a request can be processed by a route handler.

// Think of it as a gatekeeper — it decides “Can this request proceed?



// Real-World Analogy:
// Imagine an office building with a security guard at the entrance. Before letting someone in, the guard checks their ID badge.
// In NestJS, a Guard checks things like JWT tokens, roles, or permissions before the request hits your controller.



// 🧠 When is a Guard used?
// Authentication: Is the user logged in?
// Authorization: Does the user have the required role (e.g., admin)?
// Custom logic: Is this IP blacklisted? Is this feature enabled?




// 🧪 Basic Code Example – JWT Auth Guard
// auth/jwt-auth.guard.ts
import { CanActivate, ExecutionContext, Injectable } from "@nestjs/common";
import { Request } from "express";
import * as jwt from "jsonwebtoken";

@Injectable()
export class JwtAuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest<Request>();
    const token = request.headers["authorization"]?.split(" ")[1];

    if (!token) return false;

    try {
      const decoded = jwt.verify(token, "my-secret-key");
      request["user"] = decoded;
      return true;
    } catch {
      return false;
    }
  }
}




💥 How to Apply the Guard:
// users.controller.ts
import { Controller, Get, UseGuards } from '@nestjs/common';
import { JwtAuthGuard } from './auth/jwt-auth.guard';

@Controller('users')
export class UsersController {
  @Get('profile')
  @UseGuards(JwtAuthGuard)
  getProfile() {
    return { message: 'This is a protected route' };
  }
}




| Feature      | Description                           |
| ------------ | ------------------------------------- |
| Purpose      | Determine if the request can continue |
| Interface    | `CanActivate`                         |
| Used For     | Auth, role checks, request validation |
| Return       | `true` to allow, `false` to deny      |
| Applied With | `@UseGuards()` decorator              |
